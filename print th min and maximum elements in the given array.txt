import java.util.Scanner;
public class Main{
    public static int maxValue(int arr[],int size){
        int max=arr[0];
          for (int i=1;i<size;i++){
            if(max<arr[i]){
                max=arr[i];
             }
        }return max;
    }
    public static int minValue(int arr[],int size){
        int min=arr[0];
        
        for(int i=1;i<size;i++){
            if(min>arr[i]){
            min=arr[i];
            
        }
        
    }
    return min;
}

    
    public static void main(String[] args){
        Scanner obj=new Scanner(System.in);
        int size=obj.nextInt();
        int arr[]=new int[size];
        for(int i=0;i<size;i++){
            arr[i]=obj.nextInt();
        }
       System.out.println(minValue(arr,size));
       System.out.println(maxValue(arr,size));

    }
}
