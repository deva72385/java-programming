class Solution {
    public int findDelayedArrivalTime(int arrivalTime, int delayedTime) {
        int i=0;
       
        int time=arrivalTime+delayedTime;
        
        
        if(time==24){
            return 0;
        }
       else if(time>24){

            return time-24  ;
        }
        
        return time;
    }
}
